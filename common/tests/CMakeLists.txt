cmake_minimum_required(VERSION 3.20)
project(kbm_firmware_host_tests LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
option(KBM_SANITIZE "Enable host-only address/undefined sanitizers" OFF)
if(KBM_SANITIZE AND NOT MSVC)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
endif()
set(ROOT "${CMAKE_CURRENT_LIST_DIR}/../..")
add_executable(firmware-mocks firmware_mocks.cpp
    "${ROOT}/master/src/protocol.c" "${ROOT}/master/src/native_dhcp.c")
target_include_directories(firmware-mocks PRIVATE .. "${ROOT}/master/src" "${ROOT}/slave/kbm-slave/src")
if(MSVC)
    target_compile_options(firmware-mocks PRIVATE /W4 /WX /UNDEBUG)
else()
    target_compile_options(firmware-mocks PRIVATE -Wall -Wextra -Werror -UNDEBUG)
endif()
add_test(NAME firmware-mocks COMMAND firmware-mocks)
add_executable(master-protocol "${ROOT}/master/tests/protocol_test.c"
    "${ROOT}/master/src/protocol.c" "${ROOT}/master/src/master_commands.c"
    "${ROOT}/master/src/response_tracker.c")
target_include_directories(master-protocol PRIVATE .. "${ROOT}/master/src")
add_executable(slave-protocol "${ROOT}/slave/kbm-slave/tests/protocol_test.c"
    "${ROOT}/slave/kbm-slave/src/protocol.c" "${ROOT}/slave/kbm-slave/src/command.c")
target_include_directories(slave-protocol PRIVATE .. "${ROOT}/slave/kbm-slave/src")
foreach(target master-protocol slave-protocol)
    if(MSVC)
        target_compile_options(${target} PRIVATE /UNDEBUG)
    else()
        target_compile_options(${target} PRIVATE -UNDEBUG)
    endif()
    add_test(NAME ${target} COMMAND ${target})
endforeach()

# Inspect the real descriptor arrays using TinyUSB's own macros, with only the
# unique board ID mocked. No device driver or actual USB access.
if(NOT PICO_SDK_PATH)
    if(DEFINED ENV{PICO_SDK_PATH})
        set(PICO_SDK_PATH "$ENV{PICO_SDK_PATH}")
    else()
        set(PICO_SDK_PATH "${ROOT}/vendor/pico-sdk/pico-sdk")
    endif()
endif()
set(TINYUSB "${PICO_SDK_PATH}/lib/tinyusb/src")
add_executable(slave-descriptors slave_descriptors.c
    "${ROOT}/slave/kbm-slave/src/usb_descriptors.c")
target_include_directories(slave-descriptors PRIVATE stubs .. "${ROOT}/slave/kbm-slave/src" "${TINYUSB}")
target_compile_definitions(slave-descriptors PRIVATE CFG_TUSB_MCU=OPT_MCU_NONE TUP_DCD_ENDPOINT_MAX=8)
if(MSVC)
    target_compile_options(slave-descriptors PRIVATE /W4 /WX /UNDEBUG)
else()
    target_compile_options(slave-descriptors PRIVATE -Wall -Wextra -Werror -UNDEBUG)
endif()
add_test(NAME slave-descriptors COMMAND slave-descriptors)

add_executable(static-queues static_queue_test.c "${PICO_SDK_PATH}/src/common/pico_util/queue.c")
target_include_directories(static-queues PRIVATE stubs ..)
target_compile_options(static-queues PRIVATE -Wall -Wextra -Werror -UNDEBUG)
target_link_options(static-queues PRIVATE -Wl,--wrap=calloc -Wl,--wrap=free)
add_test(NAME static-queues COMMAND static-queues)
add_executable(dma-rx-cursor dma_rx_test.cpp "${ROOT}/master/src/protocol.c")
target_include_directories(dma-rx-cursor PRIVATE .. "${ROOT}/master/src")
target_compile_options(dma-rx-cursor PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME dma-rx-cursor COMMAND dma-rx-cursor)
add_executable(usb-wake usb_wake_test.cpp)
target_include_directories(usb-wake PRIVATE ..)
target_compile_options(usb-wake PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME usb-wake COMMAND usb-wake)
add_executable(hid-controls hid_controls_test.cpp)
target_include_directories(hid-controls PRIVATE ..)
target_compile_options(hid-controls PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME hid-controls COMMAND hid-controls)
add_executable(boot-session boot_session_test.cpp)
target_include_directories(boot-session PRIVATE ..)
target_compile_options(boot-session PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME boot-session COMMAND boot-session)
